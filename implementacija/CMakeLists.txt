cmake_minimum_required(VERSION 3.20)
project(NBodySimulation VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  message(FATAL_ERROR "CPU stage requires GCC or Clang with OpenMP SIMD support. On Windows use GCC/MinGW-w64.")
endif()
find_package(OpenMP REQUIRED COMPONENTS CXX)
option(NBODY_NATIVE "Optimize for the build machine (binary is less portable)" OFF)
option(NBODY_DOUBLE "Use double instead of float throughout the simulation" OFF)
option(NBODY_VECTOR_REPORT "Emit GCC vectorization diagnostics for SIMD kernel" OFF)
add_library(nbody_core src/nbody.cpp src/direct.cpp src/simd.cpp)
target_link_libraries(nbody_core PUBLIC OpenMP::OpenMP_CXX)
# OpenMP opcija je potrebna i pri povezivanju prenosivog MinGW build-a.
target_link_options(nbody_core INTERFACE "SHELL:${OpenMP_CXX_FLAGS}")
target_compile_options(nbody_core PRIVATE -Wall -Wextra -Wpedantic -fno-math-errno)
if(NBODY_DOUBLE)
  target_compile_definitions(nbody_core PUBLIC NBODY_DOUBLE)
endif()

# Skalarne petlje ostaju skalarne da bi poređenje sa SIMD-om bilo jasno.
# Zato SIMD funkcije imaju svoj prevodilački modul, bez dodatnih omotača.
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
  set_source_files_properties(src/direct.cpp PROPERTIES COMPILE_OPTIONS "-fno-tree-vectorize;-fno-tree-slp-vectorize")
  if(NBODY_VECTOR_REPORT)
    set_source_files_properties(src/simd.cpp PROPERTIES COMPILE_OPTIONS "-fopt-info-vec-optimized")
  endif()
else()
  set_source_files_properties(src/direct.cpp PROPERTIES COMPILE_OPTIONS "-fno-vectorize;-fno-slp-vectorize")
endif()
if(NBODY_NATIVE)
  target_compile_options(nbody_core PRIVATE -march=native)
endif()
add_executable(nbody_bench main.cpp)
target_link_libraries(nbody_bench PRIVATE nbody_core)
target_compile_definitions(nbody_bench PRIVATE NBODY_BUILD_TYPE="${CMAKE_BUILD_TYPE}" NBODY_NATIVE_BUILD=$<BOOL:${NBODY_NATIVE}>)
include(CTest)
if(BUILD_TESTING)
  add_executable(nbody_tests tests.cpp)
  target_link_libraries(nbody_tests PRIVATE nbody_core)
  add_test(NAME direct_numerics COMMAND nbody_tests)
  add_test(NAME benchmark_smoke COMMAND nbody_bench --n 33 --steps 2 --repeats 1 --warmup 0 --threads 2)
  add_test(NAME simulation_smoke COMMAND nbody_bench --n 17 --mode simulate --steps 10 --repeats 1 --warmup 0 --threads 2)
  add_test(NAME reject_zero_epsilon COMMAND nbody_bench --epsilon 0)
  add_test(NAME reject_negative_count COMMAND nbody_bench --n -1)
  add_test(NAME reject_zero_threads COMMAND nbody_bench --threads 0)
  set_tests_properties(reject_zero_epsilon reject_negative_count reject_zero_threads PROPERTIES WILL_FAIL TRUE)
endif()
